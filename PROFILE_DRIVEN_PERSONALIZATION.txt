```javascript
import { createContext, useState, useContext } from 'react';

const ProfileContext = createContext();

export function ProfileProvider({ children }) {
  const [profile, setProfile] = useState(() => {
    const saved = localStorage.getItem('userProfile');
    return saved ? JSON.parse(saved) : {
      fullName: '',
      role: '',
      location: '',
      linkedinUrl: '',
      services: '',
      targetMarket: ''
    };
  });

  const updateProfile = (newProfile) => {
    setProfile(newProfile);
    localStorage.setItem('userProfile', JSON.stringify(newProfile));
  };

  return (
    <ProfileContext.Provider value={{ profile, updateProfile }}>
      {children}
    </ProfileContext.Provider>
  );
}

export function useProfile() {
  return useContext(ProfileContext);
}
```

```javascript
exports.handler = async (event) => {
  const { companyName, userProfile } = JSON.parse(event.body);
  
  const client = new Anthropic();
  
  // Profile context passed to Claude
  const profileContext = `
    User Profile (for personalization):
    - Name: ${userProfile.fullName}
    - Services: ${userProfile.services}
    - Target Market: ${userProfile.targetMarket}
    - Location: ${userProfile.location}
    
    Task: When identifying pain points, focus on areas where
    the user's services could help.
  `;
  
  const researchPrompt = `
    ${profileContext}
    
    Research ${companyName}:
    
    1. COMPANY OVERVIEW
       - What do they do?
       - Size, funding, growth
       - Key markets
    
    2. HIRING SIGNALS & OPERATIONAL COMPLEXITY
       - Recent job postings
       - Team growth (indicates scaling challenges)
       - Technical hiring (need for automation?)
    
    3. INDUSTRY TRENDS
       - What's happening in their space?
       - Market growth/shifts
       - Competitive pressure
    
    4. PAIN POINTS (ALIGNED TO USER'S SERVICES)
       - What problems do they likely face?
       - Which relate to: ${userProfile.services}?
       - Why would they need help scaling/automating?
    
    5. EXECUTIVES
       - Top 3 executives
       - Their focus areas
       - Recent activities
    
    IMPORTANT: Frame pain points in terms of 
    "${userProfile.services}" expertise.
    
    Return as JSON:
    {
      company_overview: "...",
      hiring_signals: "Hiring 50 engineers (indicates ops scaling needs)",
      industry_trends: "...",
      pain_points: [
        "Operations scaling (user can automate)",
        "Team coordination (user can document SOP)",
        ...
      ],
      executives: [
        {name, title, focus}
      ],
      fit_score: 0.85,
      fit_reason: "Scaling operations - exactly what user specializes in"
    }
  `;
  
  const response = await client.messages.create({
    model: 'claude-3-5-sonnet-20241022',
    max_tokens: 2048,
    messages: [
      { role: 'user', content: researchPrompt }
    ]
  });
  
  const research = JSON.parse(response.content[0].text);
  
  return {
    statusCode: 200,
    body: JSON.stringify({
      ...research,
      personalized: true,
      user_services: userProfile.services
    })
  };
};
```

```javascript
    KEY REQUIREMENTS:
    1. Mention their pain points from research
    2. Explain how ${userProfile.services} solves those pain points
    3. Reference their company's growth/situation
    4. Personalize with their name and role

    VARIANT A (Formal, Problem-focused):
    - Tone: Professional consultant
    - Focus: Their pain points
    - Hook: "Saw you're [problem]. We help [solution]"
    
    VARIANT B (Casual, Opportunity-focused):
    - Tone: Peer-to-peer insight sharing
    - Focus: Their growth opportunity
    - Hook: "Love what you're building. Here's a thought..."
    
    VARIANT C (Data-driven, Expertise-focused):
    - Tone: Data and case studies
    - Focus: Proven approach to their problem
    - Hook: "Companies like yours typically face [X]. We reduce it by [Y]"
    
    Sign all emails with:
    --
    ${userProfile.fullName}
    ${userProfile.role}
    📍 ${userProfile.location}
    ${userProfile.linkedinUrl ? '🔗 ' + userProfile.linkedinUrl : ''}
    
    Return ONLY valid JSON:
    {
      variant_a: {
        subject: "...",
        body: "..."
      },
      variant_b: {
        subject: "...",
        body: "..."
      },
      variant_c: {
        subject: "...",
        body: "..."
      },
      opening_hook: "Specific personalization reason"
    }
  `;
  
  const response = await client.messages.create({
    model: 'claude-3-5-sonnet-20241022',
    max_tokens: 3000,
    messages: [
      { role: 'user', content: prompt }
    ]
  });
  
  const emailVariants = JSON.parse(response.content[0].text);
  
  return {
    statusCode: 200,
    body: JSON.stringify({
      success: true,
      person: personName,
      role: role,
      company: companyName,
      from: userProfile.fullName,
      emails: emailVariants
    })
  };
```

```jsx
import { useProfile } from '../utils/profileContext';
import { useState } from 'react';

export default function Agent() {
  const { profile } = useProfile();
  const [company, setCompany] = useState('');
  const [research, setResearch] = useState(null);
  const [loading, setLoading] = useState(false);

  if (!profile.fullName) {
    return (
      <div className="setup-prompt">
        <h3>⚙️ Set up your profile first</h3>
        <p>Go to Settings to add your name, services, and target market.</p>
        <p>This helps Claude tailor research and emails to YOUR business.</p>
        <button onClick={() => window.location.hash = '#settings'}>
          Go to Settings
        </button>
      </div>
    );
  }

  const handleResearch = async () => {
    setLoading(true);
    
    // Pass profile to backend
    const researchData = await fetch('/.netlify/functions/research-enhanced', {
      method: 'POST',
      body: JSON.stringify({
        companyName: company,
        userProfile: profile
      })
    }).then(r => r.json());
    
    setResearch(researchData);
    setLoading(false);
  };

  return (
    <div className="agent-container">
      <div className="agent-header">
        <h2>⚡ AI Research Agent</h2>
        <p>Personalizing research for <strong>{profile.fullName}</strong></p>
      </div>
      
      <input
        type="text"
        placeholder="Enter company name (e.g., Paystack)"
        value={company}
        onChange={(e) => setCompany(e.target.value)}
      />
      <button onClick={handleResearch} disabled={loading}>
        {loading ? 'Researching...' : 'Research & Generate Email'}
      </button>

      {research && (
        <div className="research-section">
          <h3>📊 Research Findings</h3>
          
          {/* Fit Score */}
          <div className="fit-score">
            <strong>Fit Score: {Math.round(research.fit_score * 100)}%</strong>
            <p>{research.fit_reason}</p>
          </div>
          
          <div className="research-card">
            <h4>Company Overview</h4>
            <p>{research.company_overview}</p>
          </div>
          
          <div className="research-card">
            <h4>Hiring Signals</h4>
            <p>{research.hiring_signals}</p>
          </div>
          
          <div className="research-card">
            <h4>💡 Pain Points (Aligned to Your Services)</h4>
            <ul>
              {research.pain_points.map((point, i) => (
                <li key={i}>{point}</li>
              ))}
            </ul>
            <p className="note">
              These pain points match your expertise in: {profile.services}
            </p>
          </div>
          
          <div className="research-card">
            <h4>Key Executives</h4>
            {research.executives.map((exec, i) => (
              <div key={i}>
                <strong>{exec.name}</strong> - {exec.title}
                <p>Focus: {exec.focus}</p>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Email variants would show here with personalization */}
    </div>
  );
}
```

```jsx
import { useProfile } from '../utils/profileContext';

  const [saved, setSaved] = useState(false);

    setFormData({ ...formData, [field]: value });

    updateProfile(formData);
    setSaved(true);
    setTimeout(() => setSaved(false), 3000);

    <div className="settings-container">
```
