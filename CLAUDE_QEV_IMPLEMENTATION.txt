```javascript
const Anthropic = require('@anthropic-ai/sdk');

exports.handler = async (event) => {
  const { companyName, role } = JSON.parse(event.body);
  
  const client = new Anthropic();
  
  // Step 1: Use Claude to find likely email addresses
  const prompt = `
    You are an expert at finding business decision makers' email addresses.
    
    Company: ${companyName}
    Role: ${role}
    
    Task: Find likely email addresses for people in this role at this company.
    
    Think about:
    1. What is the company domain? (e.g., @paystack.com)
    2. Who typically fills this role? (Research common names in this industry)
    3. What email formats do tech/fintech companies use?
       - firstname@domain.com
       - f.lastname@domain.com
       - firstname.lastname@domain.com
    
    Return 5 most likely emails with estimated names.
    
    Return ONLY valid JSON (no markdown):
    [
      {
        name: "Full Name",
        email: "email@domain.com",
        likelihood: 0.85
      }
    ]
  `;
  
  const response = await client.messages.create({
    model: 'claude-3-5-sonnet-20241022',
    max_tokens: 1024,
    messages: [
      { role: 'user', content: prompt }
    ]
  });
  
  let claudeEmails = [];
  try {
    const text = response.content[0].text;
    claudeEmails = JSON.parse(text);
  } catch (e) {
    console.error('Parse error:', e);
    return {
      statusCode: 400,
      body: JSON.stringify({ error: 'Failed to parse Claude response' })
    };
  }
  
  // Step 2: Verify emails with QuickEmailVerification
  const qevApiKey = process.env.QUICKEMAILVERIFICATION_API_KEY;
  
  const verifiedEmails = [];
  
  for (const item of claudeEmails) {
    try {
      const qevResponse = await fetch(
        `https://api.quickemailverification.com/v1/verify?email=${encodeURIComponent(item.email)}&apikey=${qevApiKey}`
      );
      
      const qevData = await qevResponse.json();
      
      verifiedEmails.push({
        name: item.name,
        email: item.email,
        likelihood: item.likelihood,
        verified: qevData.result === 'valid',
        safe_to_send: qevData.safe_to_send || false,
        qev_status: qevData.result,
        disposable: qevData.disposable || false,
        accept_all: qevData.accept_all || false
      });
      
    } catch (error) {
      console.error(`Verification error for ${item.email}:`, error);
      verifiedEmails.push({
        name: item.name,
        email: item.email,
        likelihood: item.likelihood,
        verified: false,
        safe_to_send: false,
        qev_status: 'error'
      });
    }
  }
  
  // Sort: verified first, then by likelihood
  verifiedEmails.sort((a, b) => {
    if (a.verified !== b.verified) return a.verified ? -1 : 1;
    return b.likelihood - a.likelihood;
  });
  
  return {
    statusCode: 200,
    body: JSON.stringify({
      success: true,
      company: companyName,
      role: role,
      emails: verifiedEmails,
      verified_count: verifiedEmails.filter(e => e.verified).length,
      total_count: verifiedEmails.length
    })
  };
};
```

2. generate-decision-maker-email.js (NEW)

```javascript
const Anthropic = require('@anthropic-ai/sdk');

exports.handler = async (event) => {
  const { companyName, personName, role, research } = JSON.parse(event.body);
  
  const client = new Anthropic();
  
  // Generate personalized email to this specific person
  const prompt = `
    You are a 25-year business development expert writing a cold email.
    
    Target:
    - Name: ${personName}
    - Role: ${role}
    - Company: ${companyName}
    
    Research findings:
    ${JSON.stringify(research, null, 2)}
    
    Task: Write 3 personalized cold email variants to ${personName}.
    
    Each email should:
    1. Reference their specific role and responsibilities
    2. Show you understand their company's situation
    3. Reference specific research findings
    4. Personalize with their name
    5. Ask for a specific action (15-min chat)
    
    Return ONLY valid JSON:
    {
      variant_a: {
        subject: "Subject line (formal tone)",
        body: "Email body..."
      },
      variant_b: {
        subject: "Subject line (casual tone)",
        body: "Email body..."
      },
      variant_c: {
        subject: "Subject line (data-driven tone)",
        body: "Email body..."
      },
      opening_line: "Specific personalization hook"
    }
  `;
  
  const response = await client.messages.create({
    model: 'claude-3-5-sonnet-20241022',
    max_tokens: 2048,
    messages: [
      { role: 'user', content: prompt }
    ]
  });
  
  let emailVariants = {};
  try {
    const text = response.content[0].text;
    emailVariants = JSON.parse(text);
  } catch (e) {
    console.error('Parse error:', e);
    return {
      statusCode: 400,
      body: JSON.stringify({ error: 'Failed to parse email variants' })
    };
  }
  
  return {
    statusCode: 200,
    body: JSON.stringify({
      success: true,
      person: personName,
      role: role,
      company: companyName,
      emails: emailVariants
    })
  };
};
```

```jsx
import { useState } from 'react';
import { apiClient } from '../utils/apiClient.js';

export default function DecisionMakers({ companyName, research }) {
  const [decisionMakers, setDecisionMakers] = useState(null);
  const [selectedPerson, setSelectedPerson] = useState(null);
  const [emailVariants, setEmailVariants] = useState(null);
  const [loading, setLoading] = useState(false);

  // Step 1: Find and verify emails for a role
  const handleFindEmails = async (role) => {
    setLoading(true);
    
    const result = await apiClient.call('/find-decision-maker-emails', {
      companyName,
      role
    });
    
    setDecisionMakers(result.emails);
    setLoading(false);
  };

  // Step 2: Generate email for specific person
  const handleSelectPerson = async (person) => {
    setSelectedPerson(person);
    setLoading(true);
    
    const result = await apiClient.call('/generate-decision-maker-email', {
      companyName,
      personName: person.name,
      role: person.role || 'Decision Maker',
      research
    });
    
    setEmailVariants(result.emails);
    setLoading(false);
  };

  return (
    <div className="decision-makers-container">
      <h3>👥 Decision Makers</h3>
      
      {/* Step 1: Show roles with email finding */}
      {!decisionMakers && !selectedPerson && (
        <div className="roles-list">
          {[
            { role: 'Chief Operating Officer', likelihood: 0.9 },
            { role: 'VP Operations', likelihood: 0.85 },
            { role: 'Head of Business Development', likelihood: 0.8 },
            { role: 'VP Engineering', likelihood: 0.75 },
            { role: 'CEO', likelihood: 0.95 }
          ].map((item, i) => (
            <div 
              key={i} 
              className="role-card"
              onClick={() => handleFindEmails(item.role)}
            >
              <div className="role-header">
                <h4>{item.role}</h4>
                <span className="confidence">
                  {Math.round(item.likelihood * 100)}% confidence
                </span>
              </div>
              <button className="find-email-btn">
                {loading ? 'Searching...' : 'Find Email'}
              </button>
            </div>
          ))}
        </div>
      )}
      
      {/* Step 2: Show found emails */}
      {decisionMakers && !selectedPerson && (
        <div className="emails-found">
          <div className="back-btn">
            <button onClick={() => setDecisionMakers(null)}>← Back</button>
          </div>
          
          <h4>Found {decisionMakers.verified_count} Verified Emails</h4>
          
          {decisionMakers.map((person, i) => (
            <div 
              key={i} 
              className={`email-card ${person.verified ? 'verified' : 'unverified'}`}
              onClick={() => handleSelectPerson(person)}
            >
              <div className="email-header">
                <div className="person-info">
                  <strong>{person.name}</strong>
                  <p className="role-text">{person.role}</p>
                </div>
                <div className="verification-badge">
                  {person.verified ? (
                    <>
                      <span className="verified-badge">✅ VERIFIED</span>
                      {person.safe_to_send && <span className="safe">Safe to send</span>}
                    </>
                  ) : (
                    <span className="unverified-badge">⚠️ Unverified</span>
                  )}
                </div>
              </div>
              
              <div className="email-display">
                <code>{person.email}</code>
              </div>
              
              <div className="likelihood">
                Likelihood: {Math.round(person.likelihood * 100)}%
              </div>
              
              {person.disposable && (
                <div className="warning">Disposable email detected</div>
              )}
              
              <button className="select-btn">
                Generate Email →
              </button>
            </div>
          ))}
        </div>
      )}
      
      {/* Step 3: Show email variants for selected person */}
      {selectedPerson && emailVariants && (
        <div className="email-variants">
          <div className="back-btn">
            <button onClick={() => {
              setSelectedPerson(null);
              setEmailVariants(null);
            }}>
              ← Back to People
            </button>
          </div>
          
          <div className="person-header">
            <h4>Email to {selectedPerson.name}</h4>
            <p>{selectedPerson.email}</p>
          </div>
          
          <div className="variants">
            {['variant_a', 'variant_b', 'variant_c'].map((variant, i) => (
              <div key={i} className="email-variant">
                <h5>
                  {variant === 'variant_a' && '📋 Formal'}
                  {variant === 'variant_b' && '💬 Casual'}
                  {variant === 'variant_c' && '📈 Data-Driven'}
                </h5>
                
                <div className="subject">
                  <strong>Subject:</strong>
                  <p>{emailVariants[variant].subject}</p>
                </div>
                
                <div className="body">
                  <strong>Body:</strong>
                  <p>{emailVariants[variant].body}</p>
                </div>
                
                <div className="actions">
                  <button onClick={() => copyEmail(emailVariants[variant].body)}>
                    📋 Copy
                  </button>
                  <button onClick={() => sendEmail(selectedPerson.email, emailVariants[variant])}>
                    📤 Send to {selectedPerson.name}
                  </button>
                </div>
              </div>
            ))}
          </div>
        </div>
      )}
    </div>
  );
}
```

```javascript
class APIClient {
  // ... existing methods ...

    return this.call('/find-decision-maker-emails', {
      companyName,
      role
    });
  }

    return this.call('/generate-decision-maker-email', {
      companyName,
      personName,
      role,
      research
    });
  }
  
  async call(endpoint, data) {
    const response = await fetch(`/.netlify/functions${endpoint}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    });
    return response.json();
  }
}
```
